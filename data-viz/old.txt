<script>
// ── Grid config ────────────────────────────────────────────────────────────
const COLS   = 18;   // cells across
const ROWS   = 26;   // cells down (length of court)
const CELL   = 10;   // px per cell
const MAX_H  = 70;   // max bar height in canvas-px
const WALL_H = 32;   // courtbox wall height in canvas-px
const GAP    = 1.2;  // gap between bars

const CW = COLS * CELL;
const CH = ROWS * CELL;

// PX_LEFT must equal WALL_H so the left wall sits flush with the grid left edge
const PX_LEFT = WALL_H; // room for left wall
const PX_RIGHT = 4;
const PY = 4;
const canvasW = PX_LEFT + CW + PX_RIGHT;
const canvasH = MAX_H + CH + WALL_H + PY * 2;
// Alias for backward compat
const PX = PX_LEFT;

// ── Color ramp ─────────────────────────────────────────────────────────────
const STOPS = [
  {t:0,    r:8,   g:20,  b:38 },
  {t:0.15, r:8,   g:52,  b:80 },
  {t:0.32, r:10,  g:100, b:85 },
  {t:0.50, r:20,  g:175, b:115},
  {t:0.67, r:195, g:218, b:50 },
  {t:0.82, r:245, g:155, b:10 },
  {t:1,    r:239, g:60,  b:60 },
];
function lerpColor(v) {
  let lo = STOPS[0], hi = STOPS[STOPS.length-1];
  for (let i = 0; i < STOPS.length-1; i++) {
    if (v >= STOPS[i].t && v <= STOPS[i+1].t) { lo=STOPS[i]; hi=STOPS[i+1]; break; }
  }
  const t = lo.t===hi.t ? 0 : (v-lo.t)/(hi.t-lo.t);
  return [
    Math.round(lo.r+(hi.r-lo.r)*t),
    Math.round(lo.g+(hi.g-lo.g)*t),
    Math.round(lo.b+(hi.b-lo.b)*t),
  ];
}
function rgb(r,g,b,a=1){ return a<1?`rgba(${r},${g},${b},${a})`:`rgb(${r},${g},${b})`; }
function darker([r,g,b],f){ return [Math.round(r*f),Math.round(g*f),Math.round(b*f)]; }

// ── Heat data ──────────────────────────────────────────────────────────────
function makeRealistic() {
  const d=[];
  for(let r=0;r<ROWS;r++){
    d[r]=[];
    for(let c=0;c<COLS;c++){
      const ny = r/(ROWS-1);
      const nx = (c/(COLS-1))*2-1;
      // Back center
      const bc = 0.95*Math.exp(-((ny-0.22)**2/0.025+(nx)**2/0.12));
      // Net corners
      const nl = 0.88*Math.exp(-((ny-0.88)**2/0.012+(nx+0.82)**2/0.03));
      const nr = 0.88*Math.exp(-((ny-0.88)**2/0.012+(nx-0.82)**2/0.03));
      // Back wall corners
      const wl = 0.72*Math.exp(-((ny-0.04)**2/0.008+(nx+0.78)**2/0.03));
      const wr = 0.72*Math.exp(-((ny-0.04)**2/0.008+(nx-0.78)**2/0.03));
      // Mid presence
      const mid= 0.30*Math.exp(-((ny-0.5)**2/0.06+(nx)**2/0.25));
      let v = bc+nl+nr+wl+wr+mid;
      v = Math.min(1,v)*0.90 + Math.random()*0.10;
      d[r][c] = Math.round(v*100)/100;
    }
  }
  return d;
}
function makeRandom(){
  return Array.from({length:ROWS},()=>Array.from({length:COLS},()=>Math.round((0.04+Math.random()*0.96)*100)/100));
}
function makeFlat(){
  return Array.from({length:ROWS},()=>Array(COLS).fill(0.02));
}

let heat = makeRealistic();

// ── Canvas ─────────────────────────────────────────────────────────────────
const canvas = document.getElementById('court-canvas');
canvas.width  = canvasW;
canvas.height = canvasH;
canvas.style.display = 'block';
canvas.style.width   = canvasW+'px';
canvas.style.height  = canvasH+'px';
// shift the iso-wrapper left so the visual center stays centered
document.getElementById('iso-wrapper').style.marginLeft = (-WALL_H/2)+'px';

const ctx = canvas.getContext('2d');

// Floor Y coordinate (top of the floor surface in canvas coords)
const floorY = PY + MAX_H;

// ── Draw ───────────────────────────────────────────────────────────────────
function draw() {
  ctx.clearRect(0, 0, canvasW, canvasH);

  // ── 1. Court box walls (drawn BEHIND bars) ─────────────────────────────
  const wx = PX, wy = floorY;
  const ww = CW, wh = CH;

  const wallFill   = 'rgba(0,229,180,0.07)';
  const wallEdge   = 'rgba(0,229,180,0.45)';
  const wallEdgeDim= 'rgba(0,229,180,0.22)';

  // ── Back wall: sits above the top edge of the floor, rises up ──
  // In canvas coords (pre-rotateX): "up" = negative Y = rises toward viewer after transform
  // Draw as a rectangle above the floor top edge
  ctx.fillStyle = wallFill;
  ctx.strokeStyle = wallEdge;
  ctx.lineWidth = 1;
  ctx.fillRect(wx, wy - WALL_H, ww, WALL_H);
  // top edge (open top — no line here)
  // sides
  ctx.beginPath();
  ctx.moveTo(wx,      wy - WALL_H); ctx.lineTo(wx,      wy);       // left edge
  ctx.moveTo(wx + ww, wy - WALL_H); ctx.lineTo(wx + ww, wy);       // right edge
  ctx.moveTo(wx,      wy);          ctx.lineTo(wx + ww, wy);       // bottom edge (floor level)
  ctx.stroke();
  // glass sheen on back wall
  const sheenB = ctx.createLinearGradient(wx, wy - WALL_H, wx, wy);
  sheenB.addColorStop(0, 'rgba(0,229,180,0.10)');
  sheenB.addColorStop(1, 'rgba(0,229,180,0.00)');
  ctx.fillStyle = sheenB;
  ctx.fillRect(wx, wy - WALL_H, ww, WALL_H);

  // ── Front wall: below the bottom edge of the floor, drops down ──
  ctx.fillStyle = wallFill;
  ctx.fillRect(wx, wy + wh, ww, WALL_H);
  ctx.strokeStyle = wallEdgeDim;
  ctx.beginPath();
  ctx.moveTo(wx,      wy + wh);          ctx.lineTo(wx + ww, wy + wh);         // top (floor level)
  ctx.moveTo(wx,      wy + wh);          ctx.lineTo(wx,      wy + wh + WALL_H); // left edge
  ctx.moveTo(wx + ww, wy + wh);          ctx.lineTo(wx + ww, wy + wh + WALL_H); // right edge
  ctx.moveTo(wx,      wy + wh + WALL_H); ctx.lineTo(wx + ww, wy + wh + WALL_H); // bottom
  ctx.stroke();

  // ── Left wall: left edge of floor, spans full court height + walls ──
  ctx.fillStyle = wallFill;
  ctx.fillRect(wx - WALL_H, wy - WALL_H, WALL_H, wh + WALL_H * 2);
  ctx.strokeStyle = wallEdge;
  ctx.beginPath();
  ctx.moveTo(wx - WALL_H, wy - WALL_H); ctx.lineTo(wx - WALL_H, wy + wh + WALL_H); // far-left edge
  ctx.moveTo(wx - WALL_H, wy - WALL_H); ctx.lineTo(wx,          wy - WALL_H);       // top edge
  ctx.moveTo(wx - WALL_H, wy + wh + WALL_H); ctx.lineTo(wx,     wy + wh + WALL_H); // bottom edge
  ctx.stroke();
  // right edge of left wall = floor left border (drawn as part of floor border)
  // glass sheen on left wall
  const sheenL = ctx.createLinearGradient(wx - WALL_H, 0, wx, 0);
  sheenL.addColorStop(0, 'rgba(0,229,180,0.00)');
  sheenL.addColorStop(1, 'rgba(0,229,180,0.06)');
  ctx.fillStyle = sheenL;
  ctx.fillRect(wx - WALL_H, wy - WALL_H, WALL_H, wh + WALL_H * 2);

  // ── Right wall: right edge of floor ──
  ctx.fillStyle = wallFill;
  ctx.fillRect(wx + ww, wy - WALL_H, WALL_H, wh + WALL_H * 2);
  ctx.strokeStyle = wallEdgeDim;
  ctx.beginPath();
  ctx.moveTo(wx + ww + WALL_H, wy - WALL_H); ctx.lineTo(wx + ww + WALL_H, wy + wh + WALL_H);
  ctx.moveTo(wx + ww,          wy - WALL_H); ctx.lineTo(wx + ww + WALL_H, wy - WALL_H);
  ctx.moveTo(wx + ww,          wy + wh + WALL_H); ctx.lineTo(wx + ww + WALL_H, wy + wh + WALL_H);
  ctx.stroke();

  // ── Floor surface ──
  ctx.fillStyle = 'rgba(0,100,80,0.10)';
  ctx.fillRect(wx, wy, ww, wh);
  // Floor border
  ctx.strokeStyle = wallEdge;
  ctx.lineWidth = 1.5;
  ctx.strokeRect(wx, wy, ww, wh);

  // ── 2. Floor grid (subtle) ─────────────────────────────────────────────
  ctx.strokeStyle = 'rgba(255,255,255,0.05)';
  ctx.lineWidth = 0.5;
  for(let r=0;r<=ROWS;r++){
    ctx.beginPath();
    ctx.moveTo(wx, wy+r*CELL);
    ctx.lineTo(wx+ww, wy+r*CELL);
    ctx.stroke();
  }
  for(let c=0;c<=COLS;c++){
    ctx.beginPath();
    ctx.moveTo(wx+c*CELL, wy);
    ctx.lineTo(wx+c*CELL, wy+wh);
    ctx.stroke();
  }

  // ── 3. Court markings ──────────────────────────────────────────────────
  // Net (horizontal center)
  ctx.strokeStyle = 'rgba(255,255,255,0.60)';
  ctx.lineWidth = 2;
  ctx.beginPath(); ctx.moveTo(wx, wy+wh/2); ctx.lineTo(wx+ww, wy+wh/2); ctx.stroke();

  // Center vertical
  ctx.strokeStyle = 'rgba(255,255,255,0.22)';
  ctx.lineWidth = 1;
  ctx.beginPath(); ctx.moveTo(wx+ww/2, wy); ctx.lineTo(wx+ww/2, wy+wh); ctx.stroke();

  // Service lines
  ctx.strokeStyle = 'rgba(255,255,255,0.18)';
  ctx.setLineDash([4,4]);
  ctx.beginPath(); ctx.moveTo(wx, wy+wh*0.28); ctx.lineTo(wx+ww, wy+wh*0.28); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(wx, wy+wh*0.72); ctx.lineTo(wx+ww, wy+wh*0.72); ctx.stroke();
  ctx.setLineDash([]);

  // ── 4. Heat bars (painter order: back row first, then forward) ─────────
  for(let r=0;r<ROWS;r++){
    for(let c=0;c<COLS;c++){
      const v = heat[r][c];
      if(v < 0.02) continue;

      const h = v * MAX_H;
      const col = lerpColor(v);
      const x = wx + c*CELL + GAP;
      const y = wy + r*CELL + GAP;       // top-left of this cell's floor tile
      const bw = CELL - GAP*2;            // bar width
      const bh = CELL - GAP*2;            // bar depth (floor footprint height)
      const barTop = y - h;               // top of the extruded bar

      // ── Front face (bottom face of bar, facing viewer) ──
      const front = darker(col, 0.48);
      ctx.fillStyle = rgb(...front);
      ctx.fillRect(x, barTop + bh, bw, h);

      // ── Right face (right side of bar) ──
      const right = darker(col, 0.32);
      ctx.fillStyle = rgb(...right);
      // thin strip on right edge
      const rw = Math.max(1, Math.round(bw * 0.18));
      ctx.fillRect(x + bw - rw, barTop + bh - bh + 1, rw, h + bh - 1);

      // ── Top face ──
      ctx.fillStyle = rgb(...col);
      ctx.fillRect(x, barTop, bw, bh);

      // Top face subtle inner highlight
      if(v > 0.5){
        ctx.fillStyle = 'rgba(255,255,255,0.10)';
        ctx.fillRect(x+1, barTop+1, bw-2, 2);
      }
    }
  }

}

// ── Stats ──────────────────────────────────────────────────────────────────
function updateStats(){
  const all = heat.flat();
  const avg = all.reduce((a,b)=>a+b,0)/all.length;
  const hot = all.filter(v=>v>=0.7).length;
  const peak = Math.max(...all);
  document.getElementById('stat-peak').innerHTML = Math.round(peak*100)+'<span>%</span>';
  document.getElementById('stat-avg').innerHTML  = Math.round(avg*100)+'<span>%</span>';
  document.getElementById('stat-hot').innerHTML  = hot+'<span> cells</span>';
}

function render(){ draw(); updateStats(); }

// ── Tooltip ────────────────────────────────────────────────────────────────
const tt=document.getElementById('tooltip');
const ttZone=document.getElementById('tt-zone');
const ttBar=document.getElementById('tt-bar');
const ttPct=document.getElementById('tt-pct');

const rowZones=['Back Wall','Back Wall','Back','Back','Back','Mid','Mid','Mid','Mid','Mid','Mid','Mid','Mid','Mid','Front','Front','Front','Front','Front','Front','Net','Net','Net','Net','Net','Net'];
const colZones=c=>c<COLS/3?'Left':c<(COLS*2/3)?'Center':'Right';

canvas.addEventListener('mousemove', e=>{
  const rect=canvas.getBoundingClientRect();
  const sx=canvas.width/rect.width;
  const sy=canvas.height/rect.height;
  const cx=(e.clientX-rect.left)*sx;
  const cy=(e.clientY-rect.top)*sy;
  const col=Math.floor((cx-PX)/CELL);
  const row=Math.floor((cy-floorY)/CELL);
  if(row>=0&&row<ROWS&&col>=0&&col<COLS){
    const v=heat[row][col];
    const [R,G,B]=lerpColor(v);
    ttZone.textContent=(rowZones[Math.min(row,rowZones.length-1)]||'Zone')+' · '+colZones(col);
    ttBar.style.width=(v*100)+'%';
    ttBar.style.background=`rgb(${R},${G},${B})`;
    ttPct.textContent='Activity: '+Math.round(v*100)+'%';
    tt.classList.add('show');
    tt.style.left=Math.min(e.clientX+16,window.innerWidth-160)+'px';
    tt.style.top=Math.max(e.clientY-60,8)+'px';
  } else {
    tt.classList.remove('show');
  }
});
canvas.addEventListener('click', e=>{
  const rect=canvas.getBoundingClientRect();
  const sx=canvas.width/rect.width;
  const sy=canvas.height/rect.height;
  const cx=(e.clientX-rect.left)*sx;
  const cy=(e.clientY-rect.top)*sy;
  const col=Math.floor((cx-PX)/CELL);
  const row=Math.floor((cy-floorY)/CELL);
  if(row>=0&&row<ROWS&&col>=0&&col<COLS){
    heat[row][col]=Math.round(Math.min(1,heat[row][col]+0.2)*100)/100;
    if(heat[row][col]>1) heat[row][col]=0.02;
    render();
  }
});
canvas.addEventListener('mouseleave',()=>tt.classList.remove('show'));

// ── Controls ───────────────────────────────────────────────────────────────
document.getElementById('btn-randomize').addEventListener('click',()=>{heat=makeRandom();render();});
document.getElementById('btn-realistic').addEventListener('click',()=>{heat=makeRealistic();render();});
document.getElementById('btn-reset').addEventListener('click',()=>{heat=makeFlat();render();});

render();
</script>
